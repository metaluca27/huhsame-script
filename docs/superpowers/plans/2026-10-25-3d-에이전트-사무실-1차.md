**Goal:** 블렌더를 설치해 클로드 코드와 MCP로 연결하고, 길모퉁이가 있는 네모 블록 받침 위 3층 「디지털 다락방」 집(층별 사무실 책상 7개, 다락 침실, 앞마당 데크 카페, 연못 정원)을 만들어 렌더 사진과 웹용 GLB로 내보낸다.

```python
"""office.blend 검사. 사용: blender office.blend --python check_office.py -- [묶음 ...]
묶음: island house deck desks export (안 주면 전부)."""
import os
import sys

import bpy

OWNERS = ["hermi", "jujuya", "luna", "nova", "pureum", "rurik", "clo"]
PREFIXES = ("island_", "house_", "attic_", "deck_", "desk_", "chair_", "seat_",
            "rest_", "sleep_", "prop_", "cam_", "light_")
GLB_LIMIT = 5 * 1024 * 1024

GROUPS = {
    "island": ["island_ground", "island_path", "island_tree_1_leaves", "island_lamp_1_bulb",
               "island_road_south", "island_road_east", "island_sidewalk_south",
               "island_crosswalk_s1", "island_crosswalk_e1", "island_pole_1", "island_wire_1",
               "island_bike_frame",
               "island_pond_water", "island_pond_stone_1", "island_pond_lily_1", "island_pond_bench"],
    "house": ["house_floor_f1", "house_floor_f3", "house_front_f1", "house_front_f3",
              "house_front_f1_door", "house_front_f1_awning1", "house_front_sign_text", "house_front_attic_gable",
              "house_front_attic_window", "attic_floor", "attic_roof_left", "attic_roof_right",
              "attic_cushion_1", "attic_cushion_7"] + [f"sleep_{o}" for o in OWNERS],
    "deck": ["deck_floor", "deck_counter", "deck_table_1_top", "deck_table_3_parasol"]
            + [f"rest_{o}" for o in OWNERS],
    "desks": [n for o in OWNERS for n in (f"desk_{o}", f"chair_{o}_seat", f"seat_{o}", f"prop_{o}_laptop")],
    "export": ["cam_main", "cam_inside", "light_sun"],
}


def main():
    argv = sys.argv[sys.argv.index("--") + 1:] if "--" in sys.argv else []
    groups = argv or list(GROUPS)
    names = set(bpy.data.objects.keys())
    errors = []
    if not bpy.data.filepath:
        errors.append("office.blend 을 열지 못함 (아직 빌드 전?)")
    for g in groups:
        for n in GROUPS[g]:
            if n not in names:
                errors.append(f"[{g}] 없음: {n}")
    for n in sorted(names):
        if not n.startswith(PREFIXES):
            errors.append(f"이름 규칙 위반: {n}")
    if "export" in groups:
        glb = os.path.join(os.path.dirname(bpy.data.filepath), "..", "web", "office.glb")
        if not os.path.exists(glb):
            errors.append("GLB 없음: web/office.glb")
        elif os.path.getsize(glb) > GLB_LIMIT:
            errors.append(f"GLB 너무 큼: {os.path.getsize(glb) / 1e6:.1f}MB (5MB 이하)")
    if errors:
        for e in errors:
            print("❌", e)
        raise RuntimeError(f"검사 실패 {len(errors)}건")
    print(f"✅ 검사 통과: {', '.join(groups)} (객체 {len(names)}개)")


main()
```

```python
"""블렌더 도형 도우미. 모든 객체는 설계 문서의 이름 규칙대로 이름 붙인다."""
import math

import bpy
from mathutils import Vector

_mats = {}


def hexc(h):
    """'#RRGGBB' → 블렌더용 선형 색."""
    h = h.lstrip("#")
    out = []
    for i in (0, 2, 4):
        c = int(h[i:i + 2], 16) / 255
        out.append(c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4)
    return tuple(out)


def reset_scene():
    """장면을 비운다. (read_factory_settings 는 MCP 애드온까지 꺼버려서 안 씀)"""
    for o in list(bpy.data.objects):
        bpy.data.objects.remove(o, do_unlink=True)
    for coll in (bpy.data.meshes, bpy.data.materials, bpy.data.curves,
                 bpy.data.cameras, bpy.data.lights):
        for d in list(coll):
            coll.remove(d)
    _mats.clear()


def mat(name, color_hex, emission=0.0, roughness=0.7):
    if name in _mats:
        return _mats[name]
    m = bpy.data.materials.new(name)
    m.use_nodes = True
    b = m.node_tree.nodes["Principled BSDF"]
    col = (*hexc(color_hex), 1.0)
    b.inputs["Base Color"].default_value = col
    b.inputs["Roughness"].default_value = roughness
    if emission:
        b.inputs["Emission Color"].default_value = col
        b.inputs["Emission Strength"].default_value = emission
    _mats[name] = m
    return m


def _finish(o, name, material, smooth=False):
    o.name = name
    o.data.name = name
    if material is not None:
        o.data.materials.clear()
        o.data.materials.append(material)
    if smooth:
        bpy.ops.object.shade_smooth()
    return o


def box(name, size, loc, material, bevel=0.0, rot=(0, 0, 0)):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc, rotation=rot)
    o = bpy.context.active_object
    o.scale = size
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    if bevel:
        mod = o.modifiers.new("bevel", "BEVEL")
        mod.width = bevel
        mod.segments = 3
    return _finish(o, name, material)


def cyl(name, radius, depth, loc, material, rot=(0, 0, 0), verts=24):
    bpy.ops.mesh.primitive_cylinder_add(vertices=verts, radius=radius, depth=depth,
                                        location=loc, rotation=rot)
    return _finish(bpy.context.active_object, name, material)


def cone(name, r1, r2, depth, loc, material, rot=(0, 0, 0), verts=24):
    bpy.ops.mesh.primitive_cone_add(vertices=verts, radius1=r1, radius2=r2, depth=depth,
                                    location=loc, rotation=rot)
    return _finish(bpy.context.active_object, name, material)


def ball(name, radius, loc, material, scale=(1, 1, 1), low=False):
    """low=True 면 각진 로우폴리 공(나무·돌), 아니면 매끈한 공."""
    if low:
        bpy.ops.mesh.primitive_ico_sphere_add(subdivisions=1, radius=radius, location=loc)
    else:
        bpy.ops.mesh.primitive_uv_sphere_add(segments=24, ring_count=12, radius=radius, location=loc)
    o = bpy.context.active_object
    o.scale = scale
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    return _finish(o, name, material, smooth=not low)


def torus(name, major, minor, loc, material, rot=(0, 0, 0)):
    bpy.ops.mesh.primitive_torus_add(major_radius=major, minor_radius=minor, location=loc,
                                     rotation=rot, major_segments=24, minor_segments=8)
    return _finish(bpy.context.active_object, name, material, smooth=True)


def prism(name, width, depth, height, loc, material):
    """박공(삼각기둥). x폭 width, y깊이 depth, 위로 height. loc = 바닥 중앙."""
    w, d = width / 2, depth / 2
    verts = [(-w, -d, 0), (w, -d, 0), (0, -d, height), (-w, d, 0), (w, d, 0), (0, d, height)]
    faces = [(0, 1, 2), (3, 5, 4), (0, 3, 4, 1), (1, 4, 5, 2), (2, 5, 3, 0)]
    me = bpy.data.meshes.new(name)
    me.from_pydata(verts, [], faces)
    me.update()
    o = bpy.data.objects.new(name, me)
    o.location = loc
    bpy.context.collection.objects.link(o)
    me.materials.append(material)
    return o


def text(name, body, loc, size, material, rot=(math.radians(90), 0, 0)):
    """세운 글자(기본: -y 쪽, 즉 카메라 쪽을 봄)."""
    bpy.ops.object.text_add(location=loc, rotation=rot)
    o = bpy.context.active_object
    o.data.body = body
    o.data.size = size
    o.data.align_x = "CENTER"
    o.data.align_y = "CENTER"
    o.data.extrude = 0.01
    bpy.ops.object.convert(target="MESH")
    return _finish(bpy.context.active_object, name, material)


def marker(name, loc):
    """캐릭터 자리 표시(빈 물체). 2·3차 웹이 이 이름으로 자리를 찾는다."""
    bpy.ops.object.empty_add(type="PLAIN_AXES", location=loc, radius=0.2)
    o = bpy.context.active_object
    o.name = name
    return o


def setup_render(width=1600, height=1000):
    s = bpy.context.scene
    engines = [e.identifier for e in s.render.bl_rna.properties["engine"].enum_items]
    s.render.engine = "BLENDER_EEVEE_NEXT" if "BLENDER_EEVEE_NEXT" in engines else "BLENDER_EEVEE"
    s.render.resolution_x = width
    s.render.resolution_y = height
    if s.world is None:
        s.world = bpy.data.worlds.new("world")
    s.world.use_nodes = True
    bg = s.world.node_tree.nodes["Background"]
    bg.inputs["Color"].default_value = (*hexc("4E4F7A"), 1.0)
    bg.inputs["Strength"].default_value = 0.8


def camera(name, loc, target, lens=40):
    bpy.ops.object.camera_add(location=loc)
    cam = bpy.context.active_object
    cam.name = name
    cam.data.lens = lens
    cam.rotation_euler = (Vector(target) - Vector(loc)).to_track_quat("-Z", "Y").to_euler()
    return cam


def light(name, kind, loc, energy, rot=(0, 0, 0), size=1.0, color="FFFFFF"):
    bpy.ops.object.light_add(type=kind, location=loc, rotation=rot)
    o = bpy.context.active_object
    o.name = name
    o.data.energy = energy
    o.data.color = hexc(color)
    if kind == "AREA":
        o.data.size = size
    return o
```

- [ ] **Step 4: 진입점 작성** — `build_office.py`

```python
"""디지털 다락방 장면 만들기.
터미널: blender --background --factory-startup --python build_office.py -- v1
MCP(열린 블렌더): exec(open(P, encoding='utf-8').read(), {'__file__': P, '__name__': '__main__'})"""
import importlib
import math
import os
import sys

import bpy

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.dirname(HERE)
if HERE not in sys.path:
    sys.path.insert(0, HERE)

import lib  # noqa: E402

PARTS = ["island", "house", "deck", "desks"]  # 파일이 있는 것만 만든다


def build_scene():
```

```python
from mathutils import Vector

from lib import ball, box, cone, cyl, mat, torus

HALF = 11.0           # 받침 반 폭
ROAD_W = 2.5          # 차도 폭 (남쪽 y -11~-8.5, 동쪽 x 8.5~11)
POND = (-5.0, -4.6)   # 연못 중심 (데크 카페 맞은편)
POND_R = 2.0


def tree(name, x, y, h=2.2):
    cyl(f"{name}_trunk", 0.12, h * 0.5, (x, y, h * 0.25), mat("trunk", "7A5A3A"), verts=8)

    px, py = POND
    cyl("island_pond_water", POND_R, 0.06, (px, py, 0.02), mat("water", "5FA8C8", roughness=0.05), verts=48)

             (px + (POND_R + 0.12) * math.cos(a), py + (POND_R + 0.12) * math.sin(a), 0.08),
             stone, scale=(1, 1, 0.6), low=True)

    for i, (dx, dy) in enumerate(((-0.6, 0.4), (0.5, -0.3), (0.1, 0.9))):

    wood = mat("bench_wood", "9B7653")
    box("island_pond_bench", (1.4, 0.4, 0.08), (px, py - POND_R - 0.8, 0.45), wood, bevel=0.02)
    for i, dx in enumerate((-0.55, 0.55)):
        box(f"island_pond_bench_leg{i + 1}", (0.08, 0.35, 0.42), (px + dx, py - POND_R - 0.8, 0.21), wood)

    wire("island_wire_1", tops[1], tops[0])
    wire("island_wire_2", tops[0], tops[2])

    box("island_bike_frame", (0.7, 0.04, 0.04), (1.6, -7.6, 0.45), bike, rot=(0, math.radians(-10), 0))
    box("island_bike_seat", (0.16, 0.08, 0.04), (1.35, -7.6, 0.62), mat("tire", "2A2A2A"))

def build():

    box("island_rim", (2 * HALF + 0.1, 2 * HALF + 0.1, 0.8), (0, 0, -0.75), mat("soil", "6B5A48"))

    lamp("island_lamp_1", -1.3, -6.5)
    lamp("island_lamp_2", 1.3, -6.5)
    street()
    pond()
```

```python
"""앞마당 데크 카페 = 쉬는 곳. 쉴 자리 rest_<이름> 7개."""
from lib import box, cone, cyl, mat, marker

DECK_C = (4.8, -4.8)
DECK_SIZE = (5.2, 4.4)
DECK_Z = 0.25                                   # 데크 윗면 높이
TABLES = [(3.3, -4.2), (5.3, -5.0), (3.6, -6.2)]
REST_ORDER = ["hermi", "jujuya", "luna", "nova", "pureum", "rurik", "clo"]


def table_set(name, x, y):
    metal = mat("metal", "3A3D4A")
    cyl(f"{name}_top", 0.45, 0.05, (x, y, DECK_Z + 0.75), mat("cafe_table", "F4EFE6"))
    cyl(f"{name}_leg", 0.05, 0.75, (x, y, DECK_Z + 0.375), metal, verts=8)
    cyl(f"{name}_parasol_pole", 0.03, 2.2, (x, y, DECK_Z + 1.1), metal, verts=8)
    cone(f"{name}_parasol", 1.0, 0.05, 0.4, (x, y, DECK_Z + 2.2), mat("parasol", "E8A0A8"), verts=8)
    spots = []
    for k, dx in enumerate((-0.7, 0.7)):
        box(f"{name}_stool_{k + 1}", (0.38, 0.38, 0.45), (x + dx, y, DECK_Z + 0.225),
            mat("stool", "C9A27A"), bevel=0.04)
        spots.append((x + dx, y, DECK_Z + 0.5))
    return spots


def build():
    cx, cy = DECK_C
    box("deck_floor", (*DECK_SIZE, DECK_Z), (cx, cy, DECK_Z / 2), mat("deck_wood", "B08A62"), bevel=0.03)
    spots = []
    for i, (x, y) in enumerate(TABLES):
        spots += table_set(f"deck_table_{i + 1}", x, y)
    box("deck_counter", (1.6, 0.5, 1.0), (6.4, -3.0, DECK_Z + 0.5), mat("counter", "8A6A4A"), bevel=0.03)
    box("deck_counter_top", (1.7, 0.6, 0.06), (6.4, -3.0, DECK_Z + 1.03), mat("cafe_table", "F4EFE6"))
    box("deck_counter_stool", (0.36, 0.36, 0.6), (6.4, -3.6, DECK_Z + 0.3), mat("stool", "C9A27A"), bevel=0.04)
    spots.append((6.4, -3.6, DECK_Z + 0.65))
    for owner, loc in zip(REST_ORDER, spots):
        marker(f"rest_{owner}", loc)
```
