function pcmToWavBlob(b64, rate) {
  const bin = atob(b64);
  const pcm = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) pcm[i] = bin.charCodeAt(i);
  const buf = new ArrayBuffer(44 + pcm.length);
  const v = new DataView(buf);
  const w = (off, s) => { for (let i = 0; i < s.length; i++) v.setUint8(off + i, s.charCodeAt(i)); };
  w(0, "RIFF"); v.setUint32(4, 36 + pcm.length, true); w(8, "WAVE");
  w(12, "fmt "); v.setUint32(16, 16, true); v.setUint16(20, 1, true); v.setUint16(22, 1, true);
  v.setUint32(24, rate, true); v.setUint32(28, rate * 2, true); v.setUint16(32, 2, true); v.setUint16(34, 16, true);
  w(36, "data"); v.setUint32(40, pcm.length, true);
  new Uint8Array(buf, 44).set(pcm);
  return new Blob([buf], { type: "audio/wav" });
}

async function getTtsBlobUrl(text, voice) {
  const res = await fetch(`${API}/voice/tts`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ text, preset: voice, tone: "calm" }),
  });
  const d = await res.json();
  if (!res.ok || !d.audio) throw new Error(d.error || "tts fail");
  return URL.createObjectURL(pcmToWavBlob(d.audio, d.rate || 24000));
}

  sr = new SR();
  sr.lang = "ko-KR";
  sr.interimResults = true;
  sr.continuous = true;
  sr.onresult = (e) => {
    let interim = "";
    for (let i = e.resultIndex; i < e.results.length; i++) {
      const t = e.results[i][0].transcript;
      if (e.results[i].isFinal) {
        if (t.trim()) { finals.push(t.trim()); vibe(60); }
      } else interim += t;
    }
    renderCaption(interim);
  };
  sr.onerror = (e) => {
    if (e.error === "not-allowed" || e.error === "service-not-allowed") {
      listening = false;
      updateListenUI();
      $("captionArea").innerHTML = '<span class="hint">마이크 권한이 필요해요. 브라우저 설정에서 허용해 주세요.<br>자막 없이도 아래 [말하기]는 쓸 수 있어요.</span>';
    }
  };
  sr.onend = () => {
    // 브라우저가 인식을 자동 종료해도, 듣기 상태면 다시 시작 (마주톡 패턴 + 연속 듣기)
    if (listening) { try { sr.start(); } catch (_) {} }
    else updateListenUI();
  };
  try { sr.start(); } catch (_) {}
}

$("listenBtn").addEventListener("click", () => {
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  if (!SR) {
    $("captionArea").innerHTML = '<span class="hint">이 브라우저는 자막 기능을 지원하지 않아요. (크롬/삼성인터넷 권장)<br>아래 [말하기]는 그대로 쓸 수 있어요.</span>';
    return;
  }
  if (listening) { listening = false; try { sr.stop(); } catch (_) {} updateListenUI(); return; }
  listening = true;
  updateListenUI();
  renderCaption("");
  startSR();
});

Task 1의 `getTtsBlobUrl` 함수 전체를 다음으로 교체:

```js
// ----- IndexedDB 오디오 캐시 -----
function idb() {
  return new Promise((res, rej) => {
    const r = indexedDB.open("najombwa", 1);
    r.onupgradeneeded = () => r.result.createObjectStore("tts", { keyPath: "key" });
    r.onsuccess = () => res(r.result);
    r.onerror = () => rej(r.error);
  });
}
async function cacheGet(key) {
  try {
    const d = await idb();
    return await new Promise((res) => {
      const q = d.transaction("tts").objectStore("tts").get(key);
      q.onsuccess = () => res(q.result || null);
      q.onerror = () => res(null);
    });
  } catch (_) { return null; }
}
async function cachePut(entry) {
  try {
    const d = await idb();
    d.transaction("tts", "readwrite").objectStore("tts").put(entry);
  } catch (_) {}
}
async function cacheTrim() {
  try {
    const d = await idb();
    const all = await new Promise((res) => {
      const q = d.transaction("tts").objectStore("tts").getAll();
      q.onsuccess = () => res(q.result || []);
      q.onerror = () => res([]);
    });
    if (all.length <= 100) return;
    const victims = all.filter((x) => !x.pinned).sort((a, b) => a.ts - b.ts).slice(0, all.length - 100);
    const tx = d.transaction("tts", "readwrite").objectStore("tts");
    victims.forEach((v) => tx.delete(v.key));
  } catch (_) {}
}

async function getTtsBlobUrl(text, voice, pinned) {
  const key = voice + ":" + text;
  const hit = await cacheGet(key);
  if (hit) {
    cachePut({ ...hit, ts: Date.now(), pinned: hit.pinned || !!pinned });
    return URL.createObjectURL(pcmToWavBlob(hit.pcm, hit.rate));
  }
  const res = await fetch(`${API}/voice/tts`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ text, preset: voice, tone: "calm" }),
  });
  const d = await res.json();
  if (!res.ok || !d.audio) throw new Error(d.error || "tts fail");
  cachePut({ key, pcm: d.audio, rate: d.rate || 24000, ts: Date.now(), pinned: !!pinned });
  cacheTrim();
  return URL.createObjectURL(pcmToWavBlob(d.audio, d.rate || 24000));
}
```
